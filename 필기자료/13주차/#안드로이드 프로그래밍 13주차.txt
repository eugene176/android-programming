cmd에서 하는법
sqlite3 naverDB //큰 dir/DB 생성
CREATE TABLE userTable (id char(4), userName char(15), email char(15), birthYear int); //큰 dir/DB에 들어갈 DB 생성
.schema userTable //테이블 구조를 볼 수 있음
INSERT INTO userTable VALUES('john', 'John Bann', 'john@naver.com', 1990);
INSERT INTO userTable VALUES('kim', 'Kim Chi', 'kim@daum.net', 1992);
.header on //보기 좋게 나옴
.mode column //줄정리까지 해서 보기 좋게 나옴


delete from 테이블명 where //테이블 행 삭제
update 테이블명 set 변경내용 where
drop table 테이블명//테이블삭제
.exit // 종료


안드로이드 프로그래밍에서 하는 법
DB 생성 삭제 - SQLite OpenHelper 클래스 상속 or 사용
update,delete,insert,select - SQLiteDatabase 클래스 상속 or 사용
커서 이동 - Cursor 인터페이스 상속 or 사용

oncreate,onupgrade 부분에서 테이블 생성

SQLiteOpenHelper를 상속받을 때 구현해줘야 하는 메소드
생성자
ex) public myDBHelper(Context context)

public void onCreate(SQLiteDatabase db)
public void onUpgrade(SQLiteDatabase db, int oldVersion, int newVersion)
